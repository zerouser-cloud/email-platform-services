```yaml
# Phase 999.17 — GitLab CI security pipeline reference template.
# See .planning/phases/999.17-devsecops-shift-left-security-tooling/.
# Copy to .gitlab-ci.yml (or include via GitLab `include:`) and customize.

# --- stages ---
stages:
  - security-fast    # secret-detection + sast (parallel-eligible, fail-fast)
  - security-deep    # container-config + container-cve + license + sbom (slower, may run after build)

# --- secret-detection (gitleaks, full repo scan vs dev-side staged-only) ---
secret-detection:
  stage: security-fast
  image: zricethezav/gitleaks:v8.30
  script:
    - gitleaks git --no-banner --redact
  rules:
    - if: $CI_PIPELINE_SOURCE == "merge_request_event"
    - if: $CI_COMMIT_BRANCH == "main"
  allow_failure: false  # secrets always block per D-07

# --- sast (Semgrep, full scan vs dev-side diff) ---
sast:
  stage: security-fast
  image: returntocorp/semgrep:1.50
  script:
    - semgrep scan --config p/typescript --config p/javascript --config p/nodejs --config p/owasp-top-ten --config p/r2c-security-audit --error
  rules:
    - if: $CI_PIPELINE_SOURCE == "merge_request_event"
    - if: $CI_COMMIT_BRANCH == "main"
  allow_failure: false  # SAST HIGH+ always blocks per D-07

# --- container-config (Trivy config — same as dev-side pre-push) ---
container-config:
  stage: security-deep
  image: aquasec/trivy:0.50
  script:
    - trivy config --severity HIGH,CRITICAL --exit-code 1 .
  rules:
    - if: $CI_PIPELINE_SOURCE == "merge_request_event"
    - if: $CI_COMMIT_BRANCH == "main"
  allow_failure: false

# --- container-cve (Trivy fs — CI ONLY per D-06; needs CVE feed ~1GB) ---
container-cve:
  stage: security-deep
  image: aquasec/trivy:0.50
  script:
    - trivy fs --severity HIGH,CRITICAL --exit-code 1 .
  rules:
    - if: $CI_PIPELINE_SOURCE == "merge_request_event"
    - if: $CI_COMMIT_BRANCH == "main"
  allow_failure: false  # CVE HIGH+ blocks per D-07

# --- license-scan (Trivy fs license — CI ONLY) ---
license-scan:
  stage: security-deep
  image: aquasec/trivy:0.50
  script:
    - trivy fs --scanners license --license-full --exit-code 1 --severity UNKNOWN,HIGH,CRITICAL .
  rules:
    - if: $CI_PIPELINE_SOURCE == "merge_request_event"
  allow_failure: false  # GPL/AGPL/SSPL blocks per D-07

# --- sbom (Syft — CI ONLY, after build) ---
sbom:
  stage: security-deep
  image: anchore/syft:v1.40
  script:
    - syft scan dir:. -o cyclonedx-json=sbom.cdx.json
  artifacts:
    paths: [sbom.cdx.json]
    expire_in: 90 days
  rules:
    - if: $CI_COMMIT_BRANCH == "main"
  allow_failure: false
```
