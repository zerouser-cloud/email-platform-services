```bash
#!/usr/bin/env bash
# Phase 999.17.1 — gitleaks fixture verify (positive expects exit 0, negative expects exit 1).
# See .planning/phases/999.17.1-quality-security-decouple-and-prove-fixtures/.
#
# Verify-mode bypasses .gitleaksignore allowlist via explicit `dir <fixture-subdir>`
# argument (D-07). Production scan path is path-blocked.

set -euo pipefail

RED='\033[0;31m'; GREEN='\033[0;32m'; NC='\033[0m'

PROJECT_ROOT="$(cd "$(dirname "${BASH_SOURCE[0]}")/../.." && pwd)"
cd "$PROJECT_ROOT"

# Source centralized image tags.
# shellcheck source=./_versions.sh
. "$(dirname "${BASH_SOURCE[0]}")/_versions.sh"

# ============================================================================
# Positive fixture: clean code, expect exit 0.
# ============================================================================
echo "[gitleaks] positive (clean) — expect exit 0..."
if ! docker run --rm -v "$(pwd):/repo" -w /repo \
  "$GITLEAKS_IMAGE" \
  dir scripts/security/fixtures/gitleaks/positive/ \
  --no-banner --exit-code 1; then
  echo -e "${RED}  FAIL${NC}: positive fixture flagged as a leak (false positive)"
  exit 1
fi
echo -e "${GREEN}  OK${NC}: gitleaks positive fixture clean."

# ============================================================================
# Negative fixture: contains AKIAQYLPMN5HEXAMPLE2 (RESEARCH §Pitfall 1).
# Expect exit 1 (gitleaks catches the leak). If exit 0, gate is fail-open.
# ============================================================================
echo "[gitleaks] negative (leak) — expect exit 1..."
if docker run --rm -v "$(pwd):/repo" -w /repo \
     "$GITLEAKS_IMAGE" \
     dir scripts/security/fixtures/gitleaks/negative/ \
     --no-banner --exit-code 1; then
  echo -e "${RED}  FAIL${NC}: gitleaks negative fixture not detected (silent fail-open!)"
  exit 1
fi
echo -e "${GREEN}  OK${NC}: gitleaks negative fixture detected."

exit 0
```
