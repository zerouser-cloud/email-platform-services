```bash
#!/usr/bin/env bash
# Phase 999.17.1 — security verify orchestrator (D-04, D-06).
# Dispatches to 6 per-gate verify scripts via declare -A associative array
# (branching-patterns skill: no switch/case, no if/elif chains).
# Aggregates exit codes; prints N/6 GREEN matrix.
#
# Per-gate scripts created in Plans 05-08:
#   - verify-gitleaks.sh    (Plan 05)
#   - verify-semgrep.sh     (Plan 06)
#   - verify-trivy-config.sh (Plan 07)
#   - verify-trivy-fs.sh    (Plan 07)
#   - verify-audit.sh       (Plan 08)
#   - verify-build-trust.sh (Plan 08)

set -euo pipefail

RED='\033[0;31m'; GREEN='\033[0;32m'; YELLOW='\033[1;33m'; NC='\033[0m'

PROJECT_ROOT="$(cd "$(dirname "${BASH_SOURCE[0]}")/../.." && pwd)"
cd "$PROJECT_ROOT"

# Helper functions (mirror scripts/check-env-parity.sh pattern).
pass() { echo -e "${GREEN}  PASS${NC}: $1"; }
fail() { echo -e "${RED}  FAIL${NC}: $1"; }
warn() { echo -e "${YELLOW}  WARN${NC}: $1"; }

echo "============================================================"
echo " Phase 999.17.1 — security verify (6 gates × positive+negative)"
echo "============================================================"

# Dispatch table — branching-patterns skill: declare -A over switch/case.
declare -A GATES=(
  [gitleaks]="bash scripts/security/verify-gitleaks.sh"
  [semgrep]="bash scripts/security/verify-semgrep.sh"
  [trivy-config]="bash scripts/security/verify-trivy-config.sh"
  [trivy-fs]="bash scripts/security/verify-trivy-fs.sh"
  [audit]="bash scripts/security/verify-audit.sh"
  [build-trust]="bash scripts/security/verify-build-trust.sh"
)

declare -A RESULTS=()

# Iterate gates; capture exit code per gate. Sequential (not parallel) because
# verify is run on phase-boundary, not pre-push critical path. Pre-push uses
# verify-if-changed.sh which only runs when configs changed.
# W3 fix per checker: GATE_ORDER is the SINGLE source of truth for both gate
# iteration order AND gate count. Adding a 7th gate requires ONE edit to this
# array (+ one entry to GATES above) — no for-loop literal duplication.
readonly GATE_ORDER=(gitleaks semgrep trivy-config trivy-fs audit build-trust)

for gate in "${GATE_ORDER[@]}"; do
  echo
  echo "------------------------------------------------------------"
  echo " Gate: $gate"
  echo "------------------------------------------------------------"
  if ${GATES[$gate]}; then
    RESULTS[$gate]="OK"
  else
    RESULTS[$gate]="FAIL($?)"
  fi
done

# Aggregate.
echo
echo "============================================================"
echo " Verify Summary"
echo "============================================================"

green_count=0
total=${#GATES[@]}

for gate in "${GATE_ORDER[@]}"; do
  if [ "${RESULTS[$gate]}" = "OK" ]; then
    pass "$gate"
    green_count=$((green_count + 1))
  else
    fail "$gate (${RESULTS[$gate]})"
  fi
done

echo
if [ "$green_count" -eq "$total" ]; then
  echo -e "${GREEN}${green_count}/${total} GREEN${NC}"
  exit 0
fi
echo -e "${RED}${green_count}/${total} GREEN${NC}"
exit 1
```

NOTE: gates are iterated in a fixed order (`for gate in gitleaks semgrep ...`) rather than `for gate in "${!GATES[@]}"` because bash associative array iteration order is not guaranteed and we want consistent output for log-diffing. The dispatch is still array-based (no switch/case) — only the iteration order is explicit.

```bash
#!/usr/bin/env bash
# Phase 999.17.1 — verify-if-changed predicate (D-01 Approach B).
# Mirrors scripts/security/audit-if-lock-changed.sh + trivy-config-if-changed.sh idiom.
#
# Triggers `pnpm run security:verify` ONLY when security-related configs/scripts
# have changed vs origin/main. Pre-push hook calls this; if no configs changed,
# verify is skipped (no overhead for normal business-code workflow).
#
# Pathspecs cover everything this phase touches:
#   - scripts/security/**       (verify scripts + fixtures)
#   - .semgrep.yml              (registry rulesets)
#   - .semgrep/rules/**         (custom rules)
#   - .gitleaks.toml            (gitleaks rule config — if used)
#   - .gitleaksignore           (path-block list)
#   - .semgrepignore            (path-block list)
#   - .trivyignore              (path-block list)

set -euo pipefail

RED='\033[0;31m'; GREEN='\033[0;32m'; YELLOW='\033[1;33m'; NC='\033[0m'

PROJECT_ROOT="$(cd "$(dirname "${BASH_SOURCE[0]}")/../.." && pwd)"
cd "$PROJECT_ROOT"

# Origin/main fail-soft fetch (mirrors audit-if-lock-changed.sh L11-18).
if ! git rev-parse origin/main >/dev/null 2>&1; then
  echo -e "${YELLOW}  WARN${NC}: origin/main not resolvable; attempting fetch."
  git fetch origin main --depth=1 >/dev/null 2>&1 || {
    echo -e "${YELLOW}  SKIP${NC}: origin/main unreachable — skipping verify-if-changed (CI is enforcement layer per 999.17 D-04)."
    exit 0
  }
fi

# Pathspecs: anything that affects security-tooling behavior.
PATHSPECS=(
  'scripts/security/**'
  '.semgrep.yml'
  '.semgrep/rules/**'
  '.gitleaks.toml'
  '.gitleaksignore'
  '.semgrepignore'
  '.trivyignore'
)

# git diff --quiet exits 0 if NO changes in pathspecs — that's our skip path.
if git diff --quiet origin/main -- "${PATHSPECS[@]}"; then
  echo -e "${GREEN}  SKIP${NC}: no security-config changes vs origin/main; verify not needed."
  exit 0
fi

echo -e "${GREEN}  RUN${NC}: security configs changed; running pnpm security:verify."
exec pnpm run security:verify
```
