**3. Update notifier main.ts** to start gRPC server alongside HTTP.

Replace `apps/notifier/src/main.ts` with the following (following parser's main.ts pattern exactly):

```typescript
import 'reflect-metadata';
import { NestFactory } from '@nestjs/core';
import { Logger } from 'nestjs-pino';
import { SERVICE, loadConfig } from '@email-platform/config';
import { NotifierEnvSchema, type NotifierEnv } from './infrastructure/config';
import { createGrpcServerOptions, SERVER, BOOTSTRAP } from '@email-platform/foundation';
import { NotifierModule } from './notifier.module';

async function bootstrap() {
  const config = loadConfig(NotifierEnvSchema) as NotifierEnv;
  const app = await NestFactory.create(NotifierModule, { bufferLogs: true });

  app.useLogger(app.get(Logger));
  app.enableShutdownHooks();

  app.connectMicroservice(createGrpcServerOptions(SERVICE.notifier, config.PROTO_DIR));

  await app.startAllMicroservices();
  await app.listen(config.NOTIFIER_PORT, SERVER.DEFAULT_HOST);
}
bootstrap().catch((err) => {
  console.error(BOOTSTRAP.FAILED_MESSAGE, err);
  process.exit(1);
});
```

Key changes from current notifier main.ts:
- Import `SERVICE, loadConfig` from `@email-platform/config` (was importing loadConfig only)
- Import `createGrpcServerOptions` from foundation
- Add `app.connectMicroservice(createGrpcServerOptions(SERVICE.notifier, config.PROTO_DIR))`
- Add `await app.startAllMicroservices()` before `app.listen()`
