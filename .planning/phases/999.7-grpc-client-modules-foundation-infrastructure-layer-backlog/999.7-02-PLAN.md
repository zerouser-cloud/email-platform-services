  {
    service: SERVICE.{service},
    clientToken: SERVICE.{service}.diToken,
    healthToken: {SERVICE_UPPER}_GRPC_HEALTH,
    grpcToken: {SERVICE_UPPER}_CLIENT_GRPC,
  },
  (grpcClient, cls, deadlineMs) => new {Service}Client(grpcClient, cls, deadlineMs),
);

@Module({})
export class {Service}ClientModule {

    return {
      module: {Service}ClientModule,
      imports: grpc.imports,
      providers: grpc.providers,
      exports: grpc.exports,
    };
  }
}
