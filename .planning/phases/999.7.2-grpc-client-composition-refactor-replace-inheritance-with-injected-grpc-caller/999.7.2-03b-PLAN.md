```ts
import type { ClientGrpc } from '@nestjs/microservices';
import { XxxProto, CommonProto } from '@email-platform/contracts';   // CommonProto if healthCheck method exists
import { SERVICE } from '@email-platform/config';
import type { GrpcCaller, CallOpts } from '@email-platform/foundation';

export class XxxClient {
  private readonly raw: XxxProto.XxxServiceClient;

  constructor(grpcClient: ClientGrpc, private readonly grpc: GrpcCaller) {
    this.raw = grpcClient.getService<XxxProto.XxxServiceClient>(
      SERVICE.<upstreamId>.grpc.serviceName,
    );
  }

  <methodName>(req: XxxProto.<Request>, opts?: CallOpts): Promise<XxxProto.<Response>> {
    return this.grpc.call('<methodName>', opts, (m) => this.raw.<methodName>(req, m));
  }
  // ... repeat for every RPC method
}
```
