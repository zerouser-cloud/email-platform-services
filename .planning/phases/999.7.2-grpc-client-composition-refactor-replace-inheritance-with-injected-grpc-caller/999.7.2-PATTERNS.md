private emitLog(
  method: string,
  startedAt: number,
  status: typeof GRPC_CLIENT_LOG.STATUS_OK | typeof GRPC_CLIENT_LOG.STATUS_ERROR,
): void {
  const fields: GrpcClientLogFields = {
    service: this.serviceName,
    method,
    duration_ms: Date.now() - startedAt,
    status,
    correlationId: this.cls.getId(),

}
