```bash
# Step 1: Capture baseline cache state

# Output should show "cache hit, replaying logs" OR fresh execution; capture hash

# Step 2: Bump ts-proto patch в package.json (e.g., "ts-proto": "^2.6.0" → "^2.6.1")
#   --- WAIT — этот bump затрагивает package.json + pnpm-lock.yaml; на каждом ENV это не automatic test;

# Differential proof recipe (embedded — НЕ requires actual ts-proto bump):
ORIG_HASH=$(sha256sum pnpm-lock.yaml | cut -d' ' -f1)
echo "" >> pnpm-lock.yaml  # append harmless trailing newline
NEW_HASH=$(sha256sum pnpm-lock.yaml | cut -d' ' -f1)
if [ "$ORIG_HASH" = "$NEW_HASH" ]; then
  echo "FAIL: pnpm-lock.yaml hash unchanged — cannot test cache invalidation"
  exit 1
fi

# Step 3: Re-run generate — MUST show cache MISS, not HIT
TURBO_OUTPUT=$(pnpm exec turbo run generate --filter=@email-platform/contracts 2>&1)
echo "$TURBO_OUTPUT"
if echo "$TURBO_OUTPUT" | grep -qE 'cache miss, executing'; then
  echo "PASS: cache invalidated by pnpm-lock.yaml change"
elif echo "$TURBO_OUTPUT" | grep -qE 'cache hit, replaying logs'; then
  echo "FAIL: cache HIT despite pnpm-lock.yaml change — F-24 input not effective"
  # Restore lockfile + exit
  git checkout pnpm-lock.yaml
  exit 1
fi

git checkout pnpm-lock.yaml
```
