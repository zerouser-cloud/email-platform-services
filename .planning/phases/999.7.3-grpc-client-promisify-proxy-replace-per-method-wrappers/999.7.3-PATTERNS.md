  [K in keyof T]: T[K] extends (...args: infer A) => Observable<infer R>
    ? (...args: A) => Promise<R>
    : T[K];
};
```

```

**Core Proxy pattern** (RESEARCH.md "Pattern 2" + Pitfall 3 RPC whitelist; pre-snapshot RPC method names from `Object.keys(raw)` to avoid wrapping `Object.prototype` methods like `toString`):

```typescript
export function promisifyGrpcClient<T extends object>(
  raw: T,
  opts: PromisifyOpts,
): Promisified<T> {
  // ClientGrpc.getService(name) returns a plain object with own keys equal to RPC
  // method names — verified in @nestjs/microservices/client/client-grpc.js:38-50.
  const rpcMethods = new Set(Object.keys(raw));

  return new Proxy(raw, {
    get(target, prop, receiver): unknown {
      if (typeof prop !== 'string' || !rpcMethods.has(prop)) {
        return Reflect.get(target, prop, receiver);
      }
      const orig = Reflect.get(target, prop, receiver) as
        | ((req: unknown, meta?: Metadata) => Observable<unknown>)
        | undefined;
      if (typeof orig !== 'function') {
        return orig;
      }

      return async (request: unknown, callOpts?: CallOpts): Promise<unknown> => {
        const deadlineMs = callOpts?.deadlineMs ?? opts.defaultDeadlineMs;
        const meta = new Metadata();
        meta.set(
          GRPC_CLIENT_DEADLINE.METADATA_HEADER,
          `${deadlineMs}${GRPC_CLIENT_DEADLINE.METADATA_UNIT_MILLISECONDS}`,
        );
        return lastValueFrom(orig.call(target, request, meta));
      };
    },
  }) as Promisified<T>;
}
```

**Header docstring pattern** (mirror the doc style from `grpc-caller.ts` lines 9-38 — multi-paragraph header with rationale, extension-point note, edge-case notes; explicitly callout D-03/D-05 "no logger now, future outer Proxy"):

```typescript
/**
 * Maps every `(...args) => Observable<R>` method on T to `(...args) => Promise<R>`.
 * Non-function properties and intentional Observable returns (future streaming) pass through.
 * ...
 */
```

**No magic values check:** `GRPC_CLIENT_DEADLINE.METADATA_HEADER` and `METADATA_UNIT_MILLISECONDS` (already constants in `clients.constants.ts:7-10`). The `Set(Object.keys(raw))` is a runtime snapshot — not a magic value. The `orig.call(target, ...)` form is the established `this`-binding pattern; mirrors `grpc-caller.ts` callback factory call shape.
