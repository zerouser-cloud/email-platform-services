const grpc = defineGrpcClient<{Upstream}Proto.{Upstream}ServiceClient>({
  service: SERVICE.{upstream},
  clientToken: SERVICE.{upstream}.diToken,
});

// Named re-exports for local consumers (health.controller.ts via barrel — D-10).

export const {UPSTREAM}_GRPC_HEALTH = grpc.healthToken;

@Module({})
export class {Upstream}ClientModule {
  static forRoot(): DynamicModule {
    return {
      module: {Upstream}ClientModule,
      imports: grpc.imports,
      providers: grpc.providers,
      exports: grpc.exports,
    };
  }
}
