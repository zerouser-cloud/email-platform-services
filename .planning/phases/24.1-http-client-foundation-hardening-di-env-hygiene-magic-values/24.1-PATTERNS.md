```typescript
// Source: abstract-http.client.ts:160-196 — moved VERBATIM into a class.
import { Injectable } from '@nestjs/common';
import { HttpErrorNormalizer } from '../errors/http-error-normalizer';
import { HttpError } from '../errors/http-error';
import type { HttpCallOpts, HttpMethod } from './types';
import type { HttpClientLogger } from '../logger/http-client-logger.port';
import { HTTP_CLIENT_LOG } from './constants';

@Injectable()
export class HttpRequestExecutor {
  constructor(
    private readonly normalizer: HttpErrorNormalizer,
    private readonly logger: HttpClientLogger,
  ) {}

  async execute(
    method: HttpMethod,
    path: string,
    init: RequestInit,
    opts: HttpCallOpts,
    baseUrl: string,
    authHeaders: Record<string, string>,
    defaultTimeoutMs: number,
    logContext: string,
  ): Promise<Response> {
    const start = Date.now();
    const url = `${baseUrl}${path}`;
    const timeoutMs = opts.timeoutMs ?? defaultTimeoutMs;
    const signal = AbortSignal.timeout(timeoutMs);
    const mergedHeaders: Record<string, string> = {
      ...authHeaders,
      ...((init.headers as Record<string, string> | undefined) ?? {}),
      ...(opts.headers ?? {}),
    };
    const finalInit: RequestInit = { ...init, method, signal, headers: mergedHeaders };
    try {
      const res = await fetch(url, finalInit);
      if (!res.ok) {
        this.logger.call({ api: logContext, method, url, duration_ms: Date.now() - start,
                            status_code: res.status, status: HTTP_CLIENT_LOG.STATUS_ERROR,
                            correlationId: undefined /* filled by adapter */ });
        throw new HttpError(res.status, url);
      }
      this.logger.call({ api: logContext, method, url, duration_ms: Date.now() - start,
                          status_code: res.status, status: HTTP_CLIENT_LOG.STATUS_OK,
                          correlationId: undefined });
      return res;
    } catch (err) {
      if (err instanceof HttpError) throw err;
      const normalized = this.normalizer.normalize(err, { url, timeoutMs });
      this.logger.call({ api: logContext, method, url, duration_ms: Date.now() - start,
                          status_code: undefined, status: HTTP_CLIENT_LOG.STATUS_ERROR,
                          correlationId: undefined });
      throw normalized;
    }
  }
}
```

```typescript
// retry-executor.port.ts
import type { HttpCallOpts, HttpMethod } from '../client/types';
import type { RetryPolicy } from '../client/types';

export interface RetryExecutorPort {
  resolvePolicy(method: HttpMethod, opts: HttpCallOpts): RetryPolicy;
  run<T>(attempt: () => Promise<T>, policy: RetryPolicy): Promise<T>;
}

// default-retry-executor.ts — lifts abstract-http.client.ts:140-158 verbatim
import { resolveRetryPolicy } from './retry-policy.resolver';
import { shouldRetry } from './retry-policy.classifier';
import { backoffWithJitter } from './retry-backoff';
import { sleep } from './sleep';

export class DefaultRetryExecutor implements RetryExecutorPort {
  resolvePolicy(method: HttpMethod, opts: HttpCallOpts): RetryPolicy {
    return resolveRetryPolicy(method, opts);
  }
  async run<T>(attempt: () => Promise<T>, policy: RetryPolicy): Promise<T> {
    let lastErr: unknown;
    for (let i = 1; i <= policy.maxAttempts; i++) {
      try { return await attempt(); }
      catch (err) {
        lastErr = err;
        if (!shouldRetry(err, i, policy)) throw err;
        await sleep(backoffWithJitter(i, policy));
      }
    }
    throw lastErr;
  }
}
```

```typescript
      resetTimeout: opts.halfOpenAfterMs,
      timeout: false,              // Pitfall 1 — do not race retry loop
      errorFilter: () => true,     // we drive open() manually
    });
    this.wireEvents();
  }

  private wrap(inner: (...a: TArgs) => Promise<TReturn>): (...a: TArgs) => Promise<TReturn> {
    return async (...args: TArgs) => {
      try {
        const r = await inner(...args);
        this.consecutiveFailures = 0;
        return r;
      } catch (err) {
        if (this.breaker.halfOpen) {
          this.consecutiveFailures = 0;
          this.breaker.open();
          throw err;
        }
        this.consecutiveFailures += 1;
        if (this.consecutiveFailures >= this.opts.consecutiveThreshold) {
          this.consecutiveFailures = 0;
          this.breaker.open();
        }
        throw err;
      }
    };
  }

  private wireEvents(): void {
    const bind = (ev: Transition): ListenerPair => {
      const fn = (): void => this.onTransition(ev);
      this.breaker.on(ev, fn);
      return [ev, fn];
    };
    this.listeners.push(bind('open'), bind('halfOpen'), bind('close'));
  }

  async fire(...args: TArgs): Promise<TReturn> {
    try { return await this.breaker.fire(...args); }
    catch (err) {
      // D-13: substring match stays private
      if (this.breaker.opened
        || (err instanceof Error && err.message.includes(CB_BREAKER_OPEN_MESSAGE_SUBSTRING))) {
        throw new CircuitOpenError(this.opts.api);
      }
      throw err;
    }
  }

  get state(): CircuitState {
    const cb = this.breaker.opened ? 'opened' : this.breaker.halfOpen ? 'halfOpen' : 'closed';
    return { cb, consecutiveFailures: this.consecutiveFailures };
  }

  dispose(): void {
    for (const [ev, fn] of this.listeners) this.breaker.off(ev, fn);
    this.breaker.shutdown();
  }
}
```

### Pattern S8: `protected override buildAuthHeaders()` in vendor subclasses

### Pattern S9: Drop `HTTP_CLIENT_` prefix when inside a subfolder

**Source:** refactor intent per CONTEXT.md D-14. `http-client.constants.ts:18` had `HTTP_CLIENT_DEFAULTS` as a flat dump. When split into `retry/constants.ts` + `circuit-breaker/constants.ts` + `errors/constants.ts`, the `HTTP_CLIENT_` prefix is redundant and noisy — use `RETRY_DEFAULTS`, `CB_DEFAULTS`, `IDEMPOTENT_METHODS`, `TIMEOUT_ERR_NAMES`, `CB_BREAKER_OPEN_MESSAGE_SUBSTRING`.

Exception: `HTTP_CLIENT_LOG` (client/constants.ts) and `HTTP_CLIENT_HEADERS` (client/request.constants.ts) stay prefixed because they're consumer-facing and the prefix helps disambiguation at import sites.
